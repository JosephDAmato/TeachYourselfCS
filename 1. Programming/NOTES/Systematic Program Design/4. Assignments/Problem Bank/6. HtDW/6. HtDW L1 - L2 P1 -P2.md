```LISP
(require 2htdp/image)
(require 2htdp/universe)

;; countdown-animation starter.rkt

; 
; PROBLEM:
; 
; Design an animation of a simple countdown. 
; 
; Your program should display a simple countdown, that starts at ten, and
; decreases by one each clock tick until it reaches zero, and stays there.
; 
; To make your countdown progress at a reasonable speed, you can use the 
; rate option to on-tick. If you say, for example, 
; (on-tick advance-countdown 1) then big-bang will wait 1 second between 
; calls to advance-countdown.
; 
; Remember to follow the HtDW recipe! Be sure to do a proper domain 
; analysis before starting to work on the code file.
; 
; Once you are finished the simple version of the program, you can improve
; it by reseting the countdown to ten when you press the spacebar.
; 
(require 2htdp/image)
(require 2htdp/universe)

;; My world program  (make this more specific)

;; =================
;; Constants:
(define HEIGHT 900)
(define WIDTH 900)
(define FONT-COLOR "Cornflowerblue")
(define FONT-SIZE 250)

(define CTR-X (/ HEIGHT 2))
(define CTR-Y (/ WIDTH 2))

(define MTS (empty-scene WIDTH HEIGHT))

;; =================
;; Data definitions:

;; Count is a Natural [0, 10]
;; Interp. Count is a number representing a countdown of Ticks
(define CD1 10)  ;; Start
(define CD2 5)   ;; Halfway
(define CD3 0)   ;; Finsihed
#;
(define (fn-for-count cd)
  (and (>= cd 0)(<= cd 10)(... cd)))

;; =================
;; Functions:

;; Count -> Count
;; start the world with (main 10)
;; 
(define (main cd)
  (big-bang cd                     ; Count
            (on-tick   tock 1)     ; Count -> Count
            (to-draw   render)     ; Count -> Image
            (on-key key-event)))   ; Count Key -> Count
;; Count -> Count
;; produce the next number in the index down to 0
(check-expect (tock 10) 9)
(check-expect (tock 0)  0)
(check-expect (tock 6)  5)

;; (define (tock cd) cd) ;STUB

(define (tock cd)
  (if (and (> cd 0)(<= cd 10))
  (- cd 1)
  cd))

;; Count -> Image
;; render a text image of Count  
(check-expect (render 10)(place-image (text (number->string 10) FONT-SIZE FONT-COLOR) CTR-X CTR-Y MTS))

;; (define (render cd) (... cd)) ;;STUB

(define (render cd)
  (if (and (>= cd 0)(<= cd 10))
      (place-image (text (number->string cd) FONT-SIZE FONT-COLOR) CTR-X CTR-Y MTS)
      (place-image (text (number->string cd) FONT-SIZE FONT-COLOR) CTR-X CTR-Y MTS)))
;; Count KeyEvent => Count
;; Produces a Count starting back at 10 when the SpaceBar is pressed
(check-expect (key-event 1 " ") 10)
(check-expect (key-event 1 "a")  1)
(check-expect (key-event 0 " ") 10)
(check-expect (key-event 0 "a")  0)

;; (define (key-event cd ke ) cd) ;STUB

(define (key-event cd ke)
  (cond
    [(key=? ke " ") 10]
    [else cd]))
```

```LISP
; 
; PROBLEM:
; 
; Design an animation of a traffic light. 
; 
; Your program should show a traffic light that is red, then green, 
; then yellow, then red etc. For this program, your changing world 
; state data definition should be an enumeration.
; 
; Here is what your program might look like if the initial world

; .
; Next:
; .
; Next:
; .
;

;

(define CTR-Y (/ WIDTH 2))

               (list
                   GREEN-OFF
                   YELLOW-OFF
                   RED-ON)
             (list(make-posn LIGHT-POSX GREEN-POSY )
                  (make-posn LIGHT-POSX YELLOW-POSY)
                  (make-posn LIGHT-POSX RED-POSY))

                   GREEN-ON
                   YELLOW-OFF

                  (make-posn LIGHT-POSX RED-POSY))

               (list
                   GREEN-OFF
                   YELLOW-ON
                   RED-OFF)
             (list(make-posn LIGHT-POSX GREEN-POSY )

                  (make-posn LIGHT-POSX RED-POSY))
                   TRAFFIC-LIGHT) CTR-X CTR-Y MTS))

;; Data definitions:

;; TrafficLight is one of the following:
;;    - "Red"
;;    - "Yellow"
;;    - "Green"

(define TL2 "Yellow")
(define TL3 "Green")
#;

  (cond [(string=? TL "Red") "Green"]
        [(string=? TL "Yellow") "Red"]
        [(string=? TL "Green") "Yellow"]))

;;  Atomic Ditinct: "Red"
;;  Atomic Distinct "Yellow"
;;  Atomic Distinct "Green"

;; =================
;; Functions:

;; tl -> tl
;; start the world with (main "Red")
;;

  (big-bang tl                   ; tl
            (on-tick   tock 2)     ; tl -> tl
            (to-draw   render)))   ; tl -> Image


;; tl -> tl
;; produce the next traffic light in sequence:
;;  - Red => Green
;;  - Green => Yellow
;;  - Yellow => Red

(check-expect (tock "Green") "Yellow")
(check-expect (tock "Red") "Green")
(check-expect (tock "Yellow") "Red")


;; (define (tock tl) tl) ;STUB

(define (tock TL)
  (cond [(string=? TL "Red") "Green"]
        [(string=? TL "Yellow") "Red"]
        [(string=? TL "Green") "Yellow"]))

;; tl -> Image

                   YELLOW-OFF

                  (make-posn LIGHT-POSX YELLOW-POSY)
                  (make-posn LIGHT-POSX RED-POSY))

               (list
                   GREEN-OFF
                   YELLOW-ON
                   RED-OFF)
             (list(make-posn LIGHT-POSX GREEN-POSY )
                  (make-posn LIGHT-POSX YELLOW-POSY)
                  (make-posn LIGHT-POSX RED-POSY))
                   TRAFFIC-LIGHT) CTR-X CTR-Y MTS))

               (list
                   GREEN-OFF
                   YELLOW-OFF
                   RED-ON)
             (list(make-posn LIGHT-POSX GREEN-POSY )
                  (make-posn LIGHT-POSX YELLOW-POSY)
                  (make-posn LIGHT-POSX RED-POSY))
```

```LISP
; 
; PROBLEM:
; 
; Use the How to Design Worlds recipe to design an interactive
; program in which a cat starts at the left edge of the display 
; and then walks across the screen to the right. When the cat
; reaches the right edge it should just keep going right off 
; the screen.
;
```

```LISP
;

(require 2htdp/image)
(require 2htdp/universe)

;; A Cat that walks left to right across the screen

;; =================
;; Constants:
(define WIDTH 600)
(define HEIGHT 400)

(define CTR-Y (/ HEIGHT 2))
(define MTS (empty-scene WIDTH HEIGHT))

(define SPEED 7)

(define CAT-IMG .)

;; =================
;; Data definitions:

;; Cat is a number
;; interp. x postion of the cat in screen coords
(define C1 0)           ; Left Edge
(define C2 (/ WIDTH 2)) ; Center
(define C3 WIDTH)       ; Right Edge
#;
(define (fn-for-cat c)
  (... c))

;; Template Rules Used:
;;  -Atomic Nondistinct: Number

;; =================
;; Functions:

;; Cat -> Cat
;; start the world with (main 0)
;; 
(define (main c)
  (big-bang c                           ; Cat
            (on-tick   advance-cat)     ; Cat -> Cat
            (to-draw   render)          ; Cat -> Image
            (on-key handle-key)         ; Cat KeyEvent -> Cat
            (on-mouse  handle-click)))  ; Cat Integer Integer MouseEvent -> Cat

;; Cat -> Cat
;; produce the next cat by advancing it SPPED px to the right
(check-expect (advance-cat 3)( + 3 SPEED))
;; (define (advance-cat c) ...)

;; Use template from Cat
(define (advance-cat c)
  (+ c SPEED))

;; Cat -> Image
;; render the cat image at appropriate x coord on MTS
(check-expect(render 4)(place-image CAT-IMG 4 CTR-Y MTS))
(define (render c)
  (place-image CAT-IMG c CTR-Y MTS))

;; Cat KeyEvent -> Cat
;; Reset cat to left edge on key press
(check-expect (handle-key 10 " ")  0) ;big bang handles space bar as " "
(check-expect (handle-key 10 "a") 10)
(check-expect (handle-key  0 " ")  0)
(check-expect (handle-key  0 "a")  0)

;; (define (handle-key c ke) 0) ; STUB

;; <Used Template from Reciepe Page>
(define( handle-key c ke)
(cond [(key=? ke " ") 0]
      [else c]))

;; Cat Mouseclick -> Cat
;; Produces a cat at the mouse click coord that moves left to right

(check-expect (handle-click 10 0 0 "button-down")0)

;; (define (handle-click c x y me) c) ;; STUB

(define (handle-click c x y me)
  (cond [(mouse=? me "button-down") x]
        [else c]))
```
