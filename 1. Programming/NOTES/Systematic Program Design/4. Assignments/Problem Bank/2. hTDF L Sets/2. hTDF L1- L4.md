;; image-area-starter.rkt

; 
; PROBLEM:
; 
; DESIGN a function called image-area that consumes an image and produces the 
; area of that image. For the area it is sufficient to just multiple the image's 
; width by its height.  Follow the HtDF recipe and leave behind commented 
; out versions of the stub and template.
; 


;image -> number
;Produces the product an image's width & height property
(check-expect (image-area (square 2 "solid" "red"))(* 2 2))
(check-expect (image-area (rectangle 10 9 "solid" "green")) 90)
;(define (image-area img) n)
;(define (image-area img)
;     (.....n)
(define (image-area img)
 (* (image-width img)(image-height img)))

```LISP
(require 2htdp/image)
;; tall-starter.rkt

; 
; PROBLEM:
; 
; DESIGN a function that consumes an image and determines whether the 
; image is tall.
; 
; Remember, when we say DESIGN, we mean follow the recipe.
; 
; Leave behind commented out versions of the stub and template.
; 

; image and number=> boolean
; Take in an image and produces a boolean (> img  tall)
(check-expect (is-tall? (rectangle 2 3 "solid" "red") 10) false)
(check-expect (is-tall? (triangle 90 "solid" "blue") 10) true) 
;(define (is-tall? img tall)
;   (...img tall)
(define (is-tall? img tall)
  (> (image-height img) tall))
```
